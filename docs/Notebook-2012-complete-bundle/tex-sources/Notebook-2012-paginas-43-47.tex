\\[0.4em]
\alpha_g &= \left(l_p \omega_e\right)^2 = l_p^2 \omega_e^2
\\[0.4em]
\alpha_g &= l_p^2 \frac{k_e}{m_e}
\approx l_p^2 \frac{F_{\max}}{a_0\,m_e}
\end{align*}
